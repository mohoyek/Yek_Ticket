import os
import subprocess
import sys
import zipfile
from datetime import datetime
from tkinter import *
from tkinter import ttk, messagebox, filedialog

# مسیر OpenSSL — تطبیق با راهنمای GICA (صفحه 3)
OPENSSL_PATH = r"C:\OpenSSL\bin\openssl.exe"

class CSRValidatorApp:
    def __init__(self, root):
        self.root = root
        self.root.title("ابزار تولید و بررسی کلیدهای GICA — نسخه نهایی")
        self.root.geometry("1000x750")
        self.root.minsize(800, 600)
        self.root.resizable(True, True)
        self.root.protocol("WM_DELETE_WINDOW", self.on_closing)

        # فونت فارسی (B Nazanin) — در صورت نبود، Tahoma
        self.font_normal = ("B Nazanin", 12)
        self.font_bold = ("B Nazanin", 12, "bold")
        self.font_small = ("B Nazanin", 10)
        try:
            test = Label(self.root, text="تست", font=self.font_normal)
            test.pack_forget()
        except:
            self.font_normal = ("Tahoma", 10)
            self.font_bold = ("Tahoma", 10, "bold")
            self.font_small = ("Tahoma", 9)

        self.generated = False
        self.user_type = "individual"  # پیش‌فرض: شخص حقیقی

        self.setup_ui()

    def setup_ui(self):
        # عنوان
        title = Label(self.root, text="🛠️ ابزار تولید و بررسی کلیدهای GICA — مطابق راهنمای رسمی", 
                      font=self.font_bold, fg="#2c3e50")
        title.pack(pady=10)

        # تب‌ها
        tab_control = ttk.Notebook(self.root)
        self.individual_tab = ttk.Frame(tab_control)
        self.ngo_tab = ttk.Frame(tab_control)
        tab_control.add(self.individual_tab, text="  شخص حقیقی  ")
        tab_control.add(self.ngo_tab, text="  شخصیت حقوقی  ")
        tab_control.pack(fill="both", expand=True, padx=15, pady=5)
        tab_control.bind("<<NotebookTabChanged>>", self.on_tab_change)

        # ساخت تب‌ها
        self.setup_individual_tab()
        self.setup_ngo_tab()

        # دکمه‌های کنترلی
        btn_frame = Frame(self.root)
        btn_frame.pack(pady=10)

        self.generate_btn = Button(btn_frame, text="تولید CSR و کلیدها", 
                                   command=self.generate, font=self.font_bold,
                                   bg="#2ecc71", fg="white", width=18, height=2)
        self.generate_btn.pack(side="left", padx=5)

        self.export_btn = Button(btn_frame, text="📦 استخراج به ZIP", 
                                 command=self.create_zip, font=self.font_bold,
                                 bg="#9b59b6", fg="white", width=18, height=2, state="disabled")
        self.export_btn.pack(side="left", padx=5)

        # بخش بررسی کلیدها (کپی/پیست)
        validate_frame = LabelFrame(self.root, text="🔍 بررسی کلیدها — محتوای فایل‌ها را در کادرها بچسبانید", 
                                    font=self.font_bold)
        validate_frame.pack(fill="both", expand=True, padx=15, pady=10)

        note_label = Label(validate_frame,
                           text="⚠️ توجه: محتوای فایل‌های CSR، کلید عمومی و خصوصی را (بدون تغییر) در کادرها بچسبانید. سپس دکمه «بررسی کلیدها» را بزنید.",
                           font=("Tahoma", 9), fg="#7f8c8d", wraplength=950, justify="center")
        note_label.pack(pady=5)

        # کادرها به صورت ریسپانسیو
        input_frame = Frame(validate_frame)
        input_frame.pack(fill="both", expand=True, padx=10, pady=10)

        labels = ["محتوای CSR", "محتوای کلید عمومی (PublicKey)", "محتوای کلید خصوصی (PrivateKey)"]
        self.text_fields = {}
        for i, label in enumerate(labels):
            col = Frame(input_frame)
            col.grid(row=0, column=i, sticky="nsew", padx=5, pady=5)
            input_frame.grid_columnconfigure(i, weight=1)

            Label(col, text=label, font=self.font_bold).pack(fill="x", pady=(0, 5))
            txt = Text(col, height=12, font=("Consolas", 10), wrap="word", bg="#f8f9fa")
            txt.pack(fill="both", expand=True)
            self.text_fields[label] = txt

        # دکمه بررسی
        Button(validate_frame, text="✅ بررسی کلیدها", 
               command=self.validate_keys_from_text, font=self.font_bold,
               bg="#3498db", fg="white", width=20, height=2).pack(pady=10)

        # لاگ
        log_frame = LabelFrame(self.root, text="وضعیت", font=self.font_bold)
        log_frame.pack(fill="both", expand=True, padx=20, pady=5)
        self.log = Text(log_frame, height=6, font=("Consolas", 9), state="disabled", bg="#f8f9fa")
        self.log.pack(fill="both", expand=True, padx=5, pady=5)

        # فوتر
        Label(self.root,
              text="✅ خروجی: CSR.txt, PrivateKey.key, PublicKey.pem | منطبق با CertProfileGuideForCSR.pdf",
              font=self.font_small, fg="#7f8c8d").pack(pady=5)

    def on_tab_change(self, event):
        tab = event.widget.tab(event.widget.select(), "text").strip()
        self.user_type = "individual" if tab == "شخص حقیقی" else "ngo"

    def setup_individual_tab(self):
        frame = self.individual_tab
        fields = [
            ("نام (به انگلیسی):", "first_name"),
            ("نام خانوادگی (به انگلیسی):", "last_name"),
            ("کد ملی (۱۰ یا ۱۱ رقم):", "national_id"),
            ("شهر (به فارسی):", "city"),
            ("استان (به فارسی):", "province"),
        ]
        self.individual_entries = {}
        for i, (label, key) in enumerate(fields):
            Label(frame, text=label, font=self.font_normal, anchor="e").grid(row=i, column=1, sticky="e", padx=10, pady=8)
            ent = Entry(frame, font=self.font_normal, width=30)
            ent.grid(row=i, column=0, sticky="w", padx=10, pady=8)
            self.individual_entries[key] = ent

    def setup_ngo_tab(self):
        frame = self.ngo_tab
        fields = [
            ("نام سازمان (به انگلیسی، بدون فاصله):", "org_name"),
            ("شناسه ملی (۱۱ رقم):", "national_id"),
            ("شهر (به فارسی):", "city"),
            ("استان (به فارسی):", "province"),
        ]
        self.ngo_entries = {}
        for i, (label, key) in enumerate(fields):
            Label(frame, text=label, font=self.font_normal, anchor="e").grid(row=i, column=1, sticky="e", padx=10, pady=8)
            ent = Entry(frame, font=self.font_normal, width=30)
            ent.grid(row=i, column=0, sticky="w", padx=10, pady=8)
            self.ngo_entries[key] = ent

        # فقط 1.OU — مطابق جدول صفحه ۱۰ و ۱۱ راهنما (OU = نامزاس مان — فارسی)
        Label(frame, text="واحد سازمانی ۱ (به فارسی — اختیاری):", font=self.font_normal, anchor="e").grid(row=4, column=1, sticky="e", padx=10, pady=8)
        self.ngo_entries["ou1"] = Entry(frame, font=self.font_normal, width=30)
        self.ngo_entries["ou1"].grid(row=4, column=0, sticky="w", padx=10, pady=8)

    def get_current_data(self):
        if self.user_type == "individual":
            data = {k: v.get().strip() for k, v in self.individual_entries.items()}
            data["type"] = "individual"
            return data
        else:
            data = {k: v.get().strip() for k, v in self.ngo_entries.items()}
            data["type"] = "ngo"
            return data

    def log_message(self, msg, level="info"):
        self.log.config(state="normal")
        prefix = {
            "info": "ℹ️ ",
            "success": "✅ ",
            "error": "❌ ",
            "warn": "⚠️ "
        }.get(level, "ℹ️ ")
        self.log.insert(END, f"{prefix}{msg}\n")
        self.log.see(END)
        self.log.config(state="disabled")

    def validate_inputs(self):
        data = self.get_current_data()
        if data["type"] == "individual":
            if not data["first_name"] or not data["last_name"]:
                messagebox.showerror("خطا", "نام و نام خانوادگی الزامی است.")
                return False
            sn = data["national_id"]
            if not sn.isdigit() or not (10 <= len(sn) <= 11):
                messagebox.showerror("خطا", "کد ملی باید ۱۰ یا ۱۱ رقمی باشد.")
                return False
        else:
            org = data["org_name"]
            if not org:
                messagebox.showerror("خطا", "نام سازمان الزامی است.")
                return False
            if not org.replace('_', '').replace('-', '').isalnum() or not org.isascii():
                messagebox.showerror("خطا", "نام سازمان باید به انگلیسی و بدون فاصله باشد (مثل TechSolutions).")
                return False
            sn = data["national_id"]
            if not sn.isdigit() or len(sn) != 11:
                messagebox.showerror("خطا", "شناسه ملی باید ۱۱ رقمی باشد.")
                return False
        if not data["city"] or not data["province"]:
            messagebox.showerror("خطا", "شهر و استان الزامی هستند.")
            return False
        return True

    def generate(self):
        if not self.validate_inputs():
            return

        self.generate_btn.config(state="disabled", text="در حال پردازش...")
        self.log_message("در حال تولید CSR و کلیدها...", "info")

        try:
            if not os.path.exists(OPENSSL_PATH):
                raise FileNotFoundError("OpenSSL در مسیر C:\\OpenSSL\\bin\\openssl.exe یافت نشد.\nلطفاً از gica.ir دانلود و نصب کنید.")

            data = self.get_current_data()
            current_dir = os.getcwd()

            # تولید CN و O مطابق راهنما (صفحه ۱۰ و ۱۱)
            if data["type"] == "individual":
                cn = f"{data['first_name']} {data['last_name']}[Sign]"  # جدول ص 10: CN = Name Family[Sign]
                org = "Unaffiliated"  # جدول ص ۱۰
                ou_lines = []
                self.export_name = f"{data['last_name']}_{data['first_name']}"
            else:
                cn = f"{data['org_name']}[Stamp]"  # جدول ص ۱۱: CN = Organization Unit[Stamp]
                org = "Non-Governmental"  # جدول ص ۱۱
                ou_lines = []
                if data.get("ou1"):
                    ou_lines.append(f"1.OU = {data['ou1']}")  # فقط 1.OU — جدول ص ۱۰ و ۱۱
                self.export_name = data["org_name"]

            # ساخت config.txt — مطابق صفحه ۴ و ۵ راهنما
            config_content = f"""[req]
prompt = no
distinguished_name = dn

[dn]
CN = {cn}
serialNumber = {data['national_id']}
O = {org}
"""
            if ou_lines:
                config_content += "\n".join(ou_lines) + "\n"
            config_content += f"""L = {data['city']}
ST = {data['province']}
C = IR
"""

            config_path = os.path.join(current_dir, "config.txt")
            with open(config_path, "w", encoding="utf-8") as f:
                f.write(config_content)
            self.log_message("✅ config.txt مطابق راهنمای GICA ساخته شد.", "success")

            # مسیرهای فایل‌ها
            self.key_path = "mykey.key"
            self.csr_path = "mycsr.txt"
            self.pub_path = "mypublickey.pem"

            # پاک‌سازی
            for f in [self.key_path, self.csr_path, self.pub_path]:
                if os.path.exists(f):
                    os.remove(f)

            # تولید CSR و کلید خصوصی — بدون رمز (-nodes) — صفحه ۶
            cmd1 = [
                OPENSSL_PATH, "req", "-new", "-utf8", "-nodes",
                "-config", config_path,
                "-newkey", "rsa:2048",
                "-keyout", self.key_path,
                "-out", self.csr_path
            ]
            result = subprocess.run(cmd1, capture_output=True, text=True, encoding="utf-8")
            if result.returncode != 0:
                raise RuntimeError(f"خطا در تولید CSR:\n{result.stderr}")

            # تولید کلید عمومی — صفحه ۷
            cmd2 = [OPENSSL_PATH, "rsa", "-in", self.key_path, "-pubout", "-out", self.pub_path]
            subprocess.run(cmd2, capture_output=True, text=True, encoding="utf-8")

            self.generated = True
            self.export_btn.config(state="normal")
            self.log_message("✅ CSR و کلیدها با موفقیت ساخته شدند.", "success")

        except Exception as e:
            self.log_message(f"❌ خطا: {e}", "error")
            messagebox.showerror("خطا", str(e))
        finally:
            self.generate_btn.config(state="normal", text="تولید CSR و کلیدها")

    def validate_keys_from_text(self):
        csr = self.text_fields["محتوای CSR"].get("1.0", END).strip()
        pub = self.text_fields["محتوای کلید عمومی (PublicKey)"].get("1.0", END).strip()
        priv = self.text_fields["محتوای کلید خصوصی (PrivateKey)"].get("1.0", END).strip()

        if not csr or not pub or not priv:
            messagebox.showwarning("اخطار", "لطفاً محتوای هر سه کادر را وارد کنید.")
            return

        self.log_message("در حال بررسی کلیدها از متن...", "info")

        temp_files = {
            "temp_csr.txt": csr,
            "temp_pub.pem": pub,
            "temp_priv.key": priv
        }

        try:
            # ذخیره موقت
            for fname, content in temp_files.items():
                with open(fname, "w", encoding="utf-8") as f:
                    f.write(content)

            # 1. بررسی کلید خصوصی
            try:
                subprocess.run([OPENSSL_PATH, "rsa", "-in", "temp_priv.key", "-check", "-noout"],
                               check=True, capture_output=True)
                self.log_message("✅ کلید خصوصی معتبر است.", "success")
            except subprocess.CalledProcessError as e:
                self.log_message("❌ کلید خصوصی نامعتبر است.", "error")
                return

            # 2. تطابق کلید عمومی و خصوصی
            try:
                mod_priv = subprocess.check_output([OPENSSL_PATH, "rsa", "-in", "temp_priv.key", "-modulus", "-noout"], text=True).strip()
                mod_pub = subprocess.check_output([OPENSSL_PATH, "rsa", "-pubin", "-in", "temp_pub.pem", "-modulus", "-noout"], text=True).strip()
                if mod_priv == mod_pub:
                    self.log_message("✅ کلید عمومی با خصوصی مطابقت دارد.", "success")
                else:
                    self.log_message("❌ کلید عمومی با خصوصی مطابقت ندارد!", "error")
                    return
            except Exception as e:
                self.log_message(f"⚠️ خطا در بررسی تطابق کلیدها: {e}", "warn")
                return

            # 3. بررسی CSR (CN و serialNumber)
            try:
                subject = subprocess.check_output([OPENSSL_PATH, "req", "-in", "temp_csr.txt", "-noout", "-subject"], text=True).strip()
                self.log_message(f"موضوع CSR: {subject}", "info")

                data = self.get_current_data()
                sn = data["national_id"]
                if data["type"] == "individual":
                    expected_cn = f"{data['first_name']} {data['last_name']}[Sign]"
                else:
                    expected_cn = f"{data['org_name']}[Stamp]"

                if f"serialNumber={sn}" not in subject:
                    self.log_message("❌ serialNumber در CSR نادرست است.", "error")
                    return
                if f"CN={expected_cn}" not in subject:
                    self.log_message(f"❌ CN مطابق انتظار نیست. (انتظار: {expected_cn})", "error")
                    return
                self.log_message("✅ CN و serialNumber در CSR صحیح هستند.", "success")

            except Exception as e:
                self.log_message(f"⚠️ خطا در بررسی CSR: {e}", "warn")
                return

            self.log_message("✅ تمام کلیدها معتبر و مطابق راهنمای GICA هستند.", "success")

        except Exception as e:
            self.log_message(f"❌ خطا در فرآیند بررسی: {e}", "error")
        finally:
            # پاک کردن فایل‌های موقت
            for fname in temp_files:
                if os.path.exists(fname):
                    os.remove(fname)

    def create_zip(self):
        if not self.generated:
            messagebox.showwarning("اخطار", "ابتدا کلیدها را تولید کنید.")
            return

        try:
            # تاریخ شمسی ساده (بدون کتابخانه)
            now = datetime.now()
            jy = now.year - 621
            if (now.month, now.day) < (3, 21):
                jy -= 1
            date_str = f"{jy:04d}-{now.month:02d}-{now.day:02d}"

            zip_name = f"{self.export_name}_{date_str}.zip"
            zip_path = filedialog.asksaveasfilename(
                title="ذخیره فایل ZIP",
                defaultextension=".zip",
                filetypes=[("ZIP files", "*.zip")],
                initialfile=zip_name
            )
            if not zip_path:
                return

            # تغییر نام فایل‌ها برای ZIP
            renames = {
                "mycsr.txt": "CSR.txt",
                "mykey.key": "PrivateKey.key",
                "mypublickey.pem": "PublicKey.pem"
            }

            with zipfile.ZipFile(zip_path, 'w') as zf:
                for src, dst in renames.items():
                    if os.path.exists(src):
                        with open(src, 'rb') as f:
                            zf.writestr(dst, f.read())
                if os.path.exists("config.txt"):
                    zf.write("config.txt", "config.txt")

            self.log_message(f"✅ فایل ZIP ذخیره شد: {os.path.basename(zip_path)}", "success")
            messagebox.showinfo("موفقیت", f"فایل‌ها در\n{zip_path}\nذخیره شدند.")

        except Exception as e:
            self.log_message(f"❌ خطا در ساخت ZIP: {e}", "error")
            messagebox.showerror("خطا", str(e))

    def on_closing(self):
        # تغییر نام فایل‌ها قبل از خروج (طبق درخواست شما)
        renames = [
            ("mycsr.txt", "CSR.txt"),
            ("mykey.key", "PrivateKey.key"),
            ("mypublickey.pem", "PublicKey.pem")
        ]
        for old, new in renames:
            if os.path.exists(old):
                try:
                    os.rename(old, new)
                    self.log_message(f"✅ فایل {old} به {new} تغییر نام یافت.", "info")
                except Exception as e:
                    self.log_message(f"⚠️ خطا در تغییر نام {old}: {e}", "warn")
        self.root.destroy()

# اجرای برنامه

    app = CSRValidatorApp(root)
